cmake_minimum_required(VERSION 3.24)
project(iiCalendarConsumer LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiAccountManager TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiFileProviderNative TRUE)
find_package(iiCalendar 0.2 CONFIG REQUIRED)
if(TARGET Qt6::Core OR TARGET iiAccountManager::iiAccountManager OR TARGET iiFileProvider::Native)
    message(FATAL_ERROR "Native iiCalendar must not import host account, file or UI SDKs")
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE iiCalendar::iiCalendar)
enable_testing()
add_test(NAME consumer COMMAND consumer)

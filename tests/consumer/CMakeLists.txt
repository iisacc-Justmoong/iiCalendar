cmake_minimum_required(VERSION 3.24)
project(iiCalendarConsumer LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiAcountManager TRUE)
find_package(iiCalendar 0.2 CONFIG REQUIRED)
if(TARGET Qt6::Core OR TARGET iiAcountManager::iiAcountManager)
    message(FATAL_ERROR "Native iiCalendar must not import the host account/UI SDK")
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE iiCalendar::iiCalendar)
enable_testing()
add_test(NAME consumer COMMAND consumer)

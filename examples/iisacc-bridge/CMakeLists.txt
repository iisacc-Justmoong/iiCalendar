cmake_minimum_required(VERSION 3.24)
# iisdk:consumer-application
project(iiCalendarIntegrationIisaccConsumer LANGUAGES CXX)
include(CTest)
find_package(iiCalendarIntegration 0.2 CONFIG REQUIRED)
find_package(iiAccountManager 0.2.9 CONFIG REQUIRED)
find_package(iiFileProviderNative 0.6 CONFIG REQUIRED)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network)
add_executable(iiCalendarIntegrationIisaccBridgeTests ../../tests/iisacc_bridge.cpp)
target_compile_features(iiCalendarIntegrationIisaccBridgeTests PRIVATE cxx_std_23)
target_link_libraries(iiCalendarIntegrationIisaccBridgeTests PRIVATE iiCalendarIntegration::iiCalendarIntegration iiAccountManager::iiAccountManager iiFileProvider::Native Qt6::Core Qt6::Network)
target_compile_definitions(iiCalendarIntegrationIisaccBridgeTests PRIVATE IISACC_ACCOUNT_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/../../resources/iisacc-account-fixture.json")
add_test(NAME iiCalendarIntegration.iisacc_bridge COMMAND iiCalendarIntegrationIisaccBridgeTests)
set_tests_properties(iiCalendarIntegration.iisacc_bridge PROPERTIES TIMEOUT 30)

cmake_minimum_required(VERSION 3.24)
# iisdk:consumer-application
project(iiCalendarFileProviderConsumer LANGUAGES CXX)
include(CTest)
find_package(iiCalendar 0.2 CONFIG REQUIRED)
find_package(iiFileProviderNative 0.6 CONFIG REQUIRED)
find_package(SQLite3 3.37 REQUIRED)
add_executable(iiCalendarFileStorageTests ../../tests/calendar_tests.cpp)
target_compile_features(iiCalendarFileStorageTests PRIVATE cxx_std_23)
target_compile_definitions(iiCalendarFileStorageTests PRIVATE IICALENDAR_FILE_PROVIDER_TEST=1)
target_include_directories(iiCalendarFileStorageTests PRIVATE App)
target_link_libraries(iiCalendarFileStorageTests PRIVATE iiCalendar::iiCalendar iiFileProvider::Native SQLite3::SQLite3)
add_test(NAME iiCalendar.file_storage COMMAND iiCalendarFileStorageTests storage)
add_executable(iiCalendarFileAccountsTests App/account_file_test.cpp)
target_compile_features(iiCalendarFileAccountsTests PRIVATE cxx_std_23)
target_link_libraries(iiCalendarFileAccountsTests PRIVATE iiCalendar::iiCalendar iiFileProvider::Native SQLite3::SQLite3)
add_test(NAME iiCalendar.file_accounts COMMAND iiCalendarFileAccountsTests)
set_tests_properties(iiCalendar.file_storage iiCalendar.file_accounts PROPERTIES
    TIMEOUT 60 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../resources")

cmake_minimum_required(VERSION 3.24)
# iisdk:consumer-application
project(iiCalendarIntegrationFileProviderConsumer LANGUAGES CXX)
include(CTest)
find_package(iiCalendarIntegration 0.2 CONFIG REQUIRED)
find_package(iiFileProviderNative 0.6 CONFIG REQUIRED)
find_package(SQLite3 3.37 REQUIRED)
add_executable(iiCalendarIntegrationFileStorageTests ../../tests/calendar_tests.cpp)
target_compile_features(iiCalendarIntegrationFileStorageTests PRIVATE cxx_std_23)
target_compile_definitions(iiCalendarIntegrationFileStorageTests PRIVATE IICALENDARINTEGRATION_FILE_PROVIDER_TEST=1)
target_include_directories(iiCalendarIntegrationFileStorageTests PRIVATE App)
target_link_libraries(iiCalendarIntegrationFileStorageTests PRIVATE iiCalendarIntegration::iiCalendarIntegration iiFileProvider::Native SQLite3::SQLite3)
add_test(NAME iiCalendarIntegration.file_storage COMMAND iiCalendarIntegrationFileStorageTests storage)
add_executable(iiCalendarIntegrationFileAccountsTests App/account_file_test.cpp)
target_compile_features(iiCalendarIntegrationFileAccountsTests PRIVATE cxx_std_23)
target_link_libraries(iiCalendarIntegrationFileAccountsTests PRIVATE iiCalendarIntegration::iiCalendarIntegration iiFileProvider::Native SQLite3::SQLite3)
add_test(NAME iiCalendarIntegration.file_accounts COMMAND iiCalendarIntegrationFileAccountsTests)
set_tests_properties(iiCalendarIntegration.file_storage iiCalendarIntegration.file_accounts PROPERTIES
    TIMEOUT 60 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../resources")
